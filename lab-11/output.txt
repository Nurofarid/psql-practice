-- 6. DELETE without WHERE removes every row (tested safely inside a transaction)
BEGIN;
BEGIN
DELETE FROM students;
DELETE 3
SELECT count(*) AS rows_left FROM students;
 rows_left 
-----------
         0
(1 row)

ROLLBACK;
ROLLBACK
SELECT count(*) AS rows_after_rollback FROM students;

---------------------
                   3
(1 row)
